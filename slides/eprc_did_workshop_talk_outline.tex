\section*{Overview}

This document drafts the main content for each slide in the DID EPRC workshop
talk. It is not the final slide deck. Each entry gives the slide-facing text and
then shows the table, equation, or figure that would appear on that slide.

The empirical motivation is Strickland, Chan, and Gottfried (2026), an
\emph{Educational Evaluation and Policy Analysis} paper on NYC's Gifted and
Talented (G\&T) program using a staggered difference-in-differences design. The
workshop exercises use synthetic panel data generated for teaching, inspired by
that setting; they are not the NYC administrative data used in the paper.

\slideheading{3}{Traditional DID Setup}

\slidecontent{In the simplest DID design, we compare two groups across two
periods. One group receives treatment after the pre-period. The other group is
not treated. DID asks whether the treated group changes more after treatment
than the comparison group changes over the same period.}

\[
\widehat{\delta}_{DID}
=
(\bar{Y}_{T,post} - \bar{Y}_{T,pre})
-
(\bar{Y}_{C,post} - \bar{Y}_{C,pre}).
\]

\slidecontent{\textbf{Concept.} DID is not just treated versus comparison after
treatment. It is the treated group's change minus the comparison group's
change.}

\begin{table}[H]
\centering
\small
\begin{tabular}{p{0.22\textwidth}p{0.18\textwidth}p{0.18\textwidth}p{0.24\textwidth}}
\toprule
\textbf{Group} & \textbf{Pre} & \textbf{Post} & \textbf{Change} \\
\midrule
Treated & Mean before year 5 & Mean after year 5 & Treated change. \\
Comparison & Mean before year 5 & Mean after year 5 & Comparison change. \\
DID & & & Treated change minus comparison change. \\
\bottomrule
\end{tabular}
\caption*{\footnotesize Planned table for Slide 3: two-group, two-period DID structure.}
\end{table}

\slideheading{4}{Parallel Trends: The Identifying Assumption}

\slidecontent{\textbf{Key identifying idea.} DID asks what would have happened
to treated units after treatment if they had not been treated. The comparison
group is useful if it tells us what would have happened to the treated group in
the absence of treatment.}

\[
E[Y_{T,post}(0) - Y_{T,pre}(0)]
=
E[Y_{C,post}(0) - Y_{C,pre}(0)].
\]

\plannedfig{simple_parallel_trends.png}{0.74}{Planned figure for Slide 4: raw group trends from the synthetic data. Treated units adopt in year 5; comparison units are never treated.}

\slideheading{5}{Classic DID Picture: Observed vs. Counterfactual}

\slidecontent{This is the classic DID teaching figure. The treated group has an
observed post-treatment outcome, but the missing object is the treated group's
counterfactual post-treatment outcome: where would the treated group have been
without treatment?}

\[
\widehat{Y}_{T,post}(0)
=
\bar{Y}_{T,pre}
+
(\bar{Y}_{C,post} - \bar{Y}_{C,pre}).
\]

\[
\widehat{\delta}_{DID}
=
\bar{Y}_{T,post}
-
\widehat{Y}_{T,post}(0).
\]

\plannedfig{classic_did_counterfactual.png}{0.74}{Planned figure for Slide 5: comparison trend, observed treated trend, dashed treated counterfactual trend, and the DID estimate.}

\slideheading{6}{DID as a Regression}

\slidecontent{The same two-by-two contrast can be estimated with a regression.
The coefficient on the interaction between treatment group and post period is
the DID estimate. Because the data are generated, we can compare the estimated
effect with the known simulated effect.}

\[
Y_{it}
=
\alpha
+
\beta Treat_i
+
\gamma Post_t
+
\delta(Treat_i \times Post_t)
+
\varepsilon_{it}.
\]

\toprule
\textbf{Term} & \textbf{Meaning} \\
\midrule
$Treat_i$ & Treated units differ from comparison units. \\
$Post_t$ & Outcomes differ after treatment begins. \\
$Treat_i \times Post_t$ & Treated units change more or less after treatment. \\
$\delta$ & DID estimate. \\
\bottomrule
\end{tabular}

\section*{Part B. TWFE and Event Studies}

\slideheading{7}{Introduce the TWFE Model}

\slidecontent{\textcolor{red}{\textbf{Introduce the TWFE model here.}} Many
applied DID papers estimate a two-way fixed effects (TWFE) model. Instead of
only using one pre-period and one post-period, TWFE uses the full panel and
controls for unit fixed effects and time fixed effects.}

\[
Y_{it} = \alpha_i + \lambda_t + \delta D_{it} + \varepsilon_{it}.
\]

\slidecontent{\textbf{Concept.} In the simple one-treatment-cohort case, TWFE
maps closely onto the familiar DID contrast. The model is attractive because it
uses repeated observations and absorbs stable unit differences and common time
shocks.}

\begin{table}[H]
\centering
\small
\begin{tabular}{p{0.27\textwidth}p{0.55\textwidth}}
\toprule
\textbf{Model piece} & \textbf{Interpretation} \\
\midrule
$\alpha_i$ & Unit fixed effects: stable differences across districts. \\
$\lambda_t$ & Year fixed effects: shocks common to all districts in a year. \\
$D_{it}$ & Treatment indicator that equals 1 after treatment begins. \\
$\delta$ & Average treatment effect in the simple setup. \\
\bottomrule
\end{tabular}
\caption*{\footnotesize Planned table for Slide 7: pieces of the TWFE model.}
\end{table}

\slideheading{8}{Event Studies: Re-centering Time Around Treatment}

\slidecontent{\textcolor{red}{\textbf{Introduce the event-study version of
TWFE here.}} An event study replaces the single treatment indicator with
indicators for time relative to treatment. This lets us look at outcomes before
and after treatment entry. In the G\&T example, event time 0 is the year a
student enters G\&T. In the synthetic data, event time 0 is year 5.}

\[
Y_{it} = \alpha_i + \lambda_t
+
\sum_{k \neq -1} \beta_k 1[t - G_i = k]
+
\varepsilon_{it}.
\]

\slidecontent{\textbf{Concept.} Pre-treatment coefficients describe whether
treated and comparison units were already diverging. Post-treatment coefficients
show how effects evolve after treatment. Event time -1 is omitted as the
reference period.}

\plannedfig{normalized_event_time_trends.png}{0.62}{Planned figure for Slide 8: outcomes centered at event time -1. Pre-treatment event times show whether treated and comparison groups were already diverging; post-treatment event times show how effects evolve after treatment.}

\slideheading{9}{Limitation of a Single TWFE Coefficient}

\slidecontent{\textcolor{red}{\textbf{Explain the limitation of TWFE: one
coefficient can hide treatment-effect dynamics.}} A single TWFE coefficient is
simple, but it compresses many possible effects into one number. This can be
misleading if effects grow, fade, or appear only several years after treatment.}

\slidecontent{In the simple design, TWFE is close to the familiar DID contrast.
Once treatment timing varies, TWFE can compare newly treated units to
never-treated units, not-yet-treated units, and already-treated units in ways
that are harder to interpret.}

\begin{table}[H]
\centering
\small
\begin{tabular}{p{0.30\textwidth}p{0.53\textwidth}}
\toprule
\textbf{What TWFE gives us} & \textbf{What it can hide} \\
\midrule
One treatment coefficient & Effects that change over time. \\
A simple average & Differences across treatment cohorts. \\
A familiar regression model & Comparisons involving already-treated units. \\
\bottomrule
\end{tabular}
\caption*{\footnotesize Planned table for Slide 9: why one TWFE coefficient can be too compressed.}
\end{table}

\section*{Part C. Staggered DID Preview}

\slideheading{10}{Staggered DID Example: G\&T Entry at Different Times}

\slidecontent{\textcolor{red}{\textbf{Use G\&T to introduce staggered treatment
timing.}} In many education settings, treatment does not start for everyone at
the same time. Students may enter G\&T at different grade points. Districts or
schools may adopt a policy in different years.}

\slidecontent{In the synthetic data, one treated group adopts in year 5 and
another adopts in year 8. A third group is never treated. This mirrors the core
staggered-adoption problem: treatment timing is itself part of the design.}

\plannedfig{staggered_timing_plot.png}{0.72}{Planned figure for Slide 10: early adopters, later adopters, and never-treated units.}

\clearpage

\slideheading{11}{Why Staggered Timing Complicates TWFE}

\slidecontent{\textcolor{red}{\textbf{Explain the problem of TWFE under
staggered adoption.}} With staggered timing, the comparison group changes over
time. In some years, later-treated units can serve as comparisons for
early-treated units. After the later group becomes treated, those comparisons
become more complicated.}

\plannedfig{staggered_group_trends.png}{0.70}{Planned figure for Slide 11: early-treated, later-treated, and never-treated group trends.}

\begin{table}[H]
\centering
\small
\begin{tabular}{p{0.30\textwidth}p{0.52\textwidth}}
\toprule
\textbf{Comparison type} & \textbf{Interpretation} \\
\midrule
Treated vs. never treated & Usually the most intuitive DID comparison. \\
Early treated vs. not-yet treated & Useful before the later group adopts. \\
Later treated vs. already treated & Can be problematic if effects evolve over time. \\
\bottomrule
\end{tabular}
\caption*{\footnotesize Planned table for Slide 11: why staggered timing changes the comparison problem.}
\end{table}

\slidecontent{This motivates the Goodman-Bacon decomposition: TWFE with
staggered adoption can be understood as a weighted average of many two-by-two
DID comparisons, not all of which answer the same policy question.}

\clearpage

\slideheading{12}{Preview of Staggered DID Estimators}

\slidecontent{\textcolor{red}{\textbf{Preview CSA logic: estimate group-time
effects, then aggregate.}} Modern staggered DID estimators avoid relying on one
opaque TWFE coefficient. Instead, they estimate treatment effects for specific
groups at specific times.}

\[
ATT(g,t)
=
E[Y_t(g) - Y_t(0) \mid G_i = g].
\]

\slidecontent{For group $g$, first treated in period $g$, estimate the effect at
time $t$ by comparing that group to appropriate untreated or not-yet-treated
units. Then aggregate group-time effects into the summary needed for the
research question.}

\[
ATT
=
\sum_{g,t} w_{gt} ATT(g,t).
\]

\begin{table}[H]
\centering
\small
\begin{tabular}{p{0.28\textwidth}p{0.54\textwidth}}
\toprule
\textbf{Estimand} & \textbf{Meaning} \\
\midrule
$ATT(g,t)$ & Effect for the cohort first treated in period $g$ at time $t$. \\
Simple aggregation & Average over group-time effects. \\
Group aggregation & Average effects by treatment cohort. \\
Event-time aggregation & Average effects by time since treatment. \\
\bottomrule
\end{tabular}
\caption*{\footnotesize Planned table for Slide 12: CSA-style group-time estimands and aggregations.}
\end{table}
