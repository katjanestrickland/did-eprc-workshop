\begin{frame}[t]{DiD Example: Gifted and Talented Programs}
\small
\setlength{\parskip}{2pt}
\textbf{Research question:} Does participation in New York City's Gifted and
Talented (G\&T) program improve student attendance and academic achievement?
\par\vspace{0.6em}
\textbf{Design:} Longitudinal student-level data and staggered DiD, following
students who entered G\&T in Grades 1--3.
\par\vspace{0.6em}
\textbf{Main findings:}
\begin{itemize}
  \setlength{\itemsep}{0.35em}
  \item Modest, sustained improvements in attendance and academic achievement,
        including middle-school mathematics and English Language Arts proficiency.
  \item Benefits accumulated over time, with effects reaching 0.11--0.18 standard
        deviations for students entering in Grade 3.
\end{itemize}
\vfill
{\scriptsize\color{MidGray}Strickland, K., Chan, W., \& Gottfried, M. A. (2026).
\textit{Longitudinal Effects of NYC's Gifted and Talented Program: Evidence From
a Staggered Difference-in-Differences Approach.} Educational Evaluation and Policy Analysis.
\href{https://doi.org/10.3102/01623737261438146}{doi:10.3102/01623737261438146}\par}
\end{frame}

\begin{frame}{The TWFE Model}
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{2pt}
\accentline
\par
TWFE is widely used to estimate treatment effects.
\[
Y_{it}
=
\alpha_i
+ \lambda_t
+ \beta D_{it}
+ u_{it}
\]
\begin{center}
\small
\begin{tabular}{L{0.22\textwidth}L{0.58\textwidth}}
\toprule
\textbf{Model Piece} & \textbf{Interpretation} \\
\midrule
$\alpha_i$ & Unit fixed effects \\
$\lambda_t$ & Time fixed effects \\
$D_{it}$ & Treatment indicator, equal to 1 if unit $i$ is treated at time $t$, and 0 otherwise \\
$\beta$ & Treatment effect \\
$u_{it}$ & Error term \\
\bottomrule
\end{tabular}
\end{center}
The coefficient from the TWFE model equals
\[
\beta
= \left(\overline Y_{G,\mathrm{post}}-\overline Y_{G,\mathrm{pre}}\right)
- \left(\overline Y_{C,\mathrm{post}}-\overline Y_{C,\mathrm{pre}}\right)
\]
\end{frame}

\vspace{0.3em}
TWFE combines multiple $2\times2$ DiD comparisons (Goodman-Bacon, 2021).
\begin{itemize}
  \setlength{\itemsep}{0.4em}
  \item Some comparisons use already-treated units as controls for
        later-treated units.
  \item If treatment effects change over time, these comparisons can distort
        the TWFE estimate.
\end{itemize}
\end{frame}

\begin{frame}[t]{Staggered DiD Estimator}
\setlength{\parskip}{2pt}
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
Callaway and Sant'Anna (2021) estimate treatment effects separately for each
cohort and period.
\par\vspace{0.25em}
{\color{UFBlue}\textbf{Group-Time ATT:}}
\[
\ATT(g,t)=\E\!\left[Y_t(g)-Y_t(0)\mid G_i=g\right]
\]
\par\vspace{0.25em}
{\color{UFBlue}\textbf{Estimation:}}
\[
\widehat{\ATT}(g,t)
=\left(\overline Y_{g,t}-\overline Y_{g,g-1}\right)
-\left(\overline Y_{C,t}-\overline Y_{C,g-1}\right)
\]
\par\vspace{0.25em}
\textbf{Example:} For units first treated in Period 5, we compare outcome
changes from Period 4 to Period 7 with those of never-treated units.
\end{frame}

\begin{frame}[t]{Weighted Averages: Summarizing Treatment Effects}
\setlength{\parskip}{2pt}
After estimating $\ATT(g,t)$, we can aggregate treatment effects to answer
different research questions.
\par\vspace{0.35em}
\begingroup
\small
\setlength{\parskip}{0pt}
\setlength{\aboverulesep}{1pt}
\setlength{\belowrulesep}{1.5pt}
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{@{}L{0.40\textwidth}@{\hspace{0.025\textwidth}}L{0.575\textwidth}@{}}
\toprule
\textbf{Method \& Research Question} & \textbf{How Effects Are Averaged} \\
\midrule
\textbf{Simple ATT}\par
What is the overall average treatment effect? &
Average all post-treatment $\ATT(g,t)$ values. Cohorts with more units or
more post-treatment periods receive greater weight. \\
\midrule
\textbf{Group ATT}\par
How do treatment effects differ across cohorts? &
First average treatment effects over time within each cohort. Then combine
the cohort averages, weighting by the number of units in each cohort. \\
\midrule
\textbf{Dynamic ATT}\par
How do treatment effects evolve over time since treatment? &
Average effects across cohorts at the same event time ($e=t-g$), weighted
by cohort size. \\
\midrule
\textbf{Calendar ATT}\par
How do treatment effects vary across calendar periods? &
Average effects across treated cohorts in the same calendar period,
weighted by cohort size. \\
\bottomrule
\end{tabular}
\endgroup
\end{frame}
